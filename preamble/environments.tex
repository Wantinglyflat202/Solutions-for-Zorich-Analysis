% exercisebody is an internal serial counter: every exercise gets a unique
% hyperlink target, even if manual display numbers are repeated accidentally.
% The public exercise counter tracks ONLY automatically numbered exercises.
\newcounter{exercise}[section]
\newtheorem{exercisebody}{习题}[section]
\renewcommand{\theexercisebody}{\thesection.\arabic{exercise}}
\NewDocumentEnvironment{exercise}{o}{%
  \IfNoValueTF{#1}
    {\stepcounter{exercise}}
    {\renewcommand{\theexercisebody}{\thesection.#1}}%
  \begin{exercisebody}%
}{\end{exercisebody}}
