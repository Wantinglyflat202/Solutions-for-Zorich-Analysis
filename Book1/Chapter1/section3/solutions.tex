\begin{exercise}
    定义了关系的复合，证明关系性质
    \label{def:b1-c1-s3-q1}
    \begin{align*}
        &\mathcal{R}_1 \circ \mathcal{R}_2:=\{(x,z)|\exists y(x\mathcal{R}_1y)\land (y\mathcal{R}_2 z)\}\\  
        &\Delta_X \text{是集合}X^2\text{的对角线}   
    \end{align*}
    证明:
    \begin{align*}
        &a):\\
        &\text{若} \mathcal{R}_1\subset X\times Y,\mathcal{R}_2\subset Y\times X\text{满足}(\mathcal{R}_2\circ\mathcal{R}_1=\Delta_X)\land(\mathcal{R}_1\circ\mathcal{R}_2=\Delta_Y)\\
        &\text{则它们都是函数关系并且给出了集合X,Y之间的互逆映射}\\
        &b):\\
        &\mathcal{R}\subset X^2,\text{请证明}\mathcal{R}\text{的传递性等价于}\mathcal{R}\circ\mathcal{R}\subset\mathcal{R}\\
        &c):\\
        &\text{如果关系}\mathcal{R}^\prime\subset Y\times X\text{和}\mathcal{R}\subset X\times Y\text{满足}(y\mathcal{R}^\prime x)\Leftrightarrow (x\mathcal{R}y),\text{则称}\mathcal{R}^\prime\text{为}\mathcal{R}\text{的转置关系}\\
        &\text{请证明}\mathcal{R}\subset X^2\text{的反对称性等价于}\mathcal{R}\cap \mathcal{R}^\prime \subset \Delta_X\\
        &d):\\
        &\text{请验证:当且仅当}\mathcal{R}\cup \mathcal{R}^\prime=X^2\text{时。集合X的任意两个元素由}\mathcal{R}\subset X^2\text{相联系（按照任意顺序）}
    \end{align*}
\end{exercise}
\begin{solution}.\\
a):\\
\begin{align*}
    &\forall x_1\in X,(x_1,x_1)\in \Delta_X\\
    \Leftrightarrow&\exists y_0\in Y,(x_1,y_0)\in \mathcal{R}_1\land (y_0,x_1)\in \mathcal{R}_2\\
    &\text{若}\exists y_1\neq y_0,(x_1,y_1)\in \mathcal{R}_1\\
    \Rightarrow&(y_0,y_1)\in \mathcal{R}_1\circ\mathcal{R}_2\Rightarrow (y_0,y_1)\in \Delta_Y\Rightarrow y_0=y_1\text{导出矛盾}
\end{align*}
因此可以证明这两个关系均为函数关系，自然地，也就给出了一组互逆映射\\
b):\\
\begin{align*}
\Rightarrow&:\\
&(x,y)\in \mathcal{R}\land (y,z)\in \mathcal{R}\text{则}(x,z)\in \mathcal{R}\\
\Rightarrow&\forall (x,y)\in \mathcal{R}\circ\mathcal{R},\exists z\in X,(x,z)\in \mathcal{R}\land (z,y)\in \mathcal{R}\\
\Rightarrow&\text{由传递性}, (x,y)\in \mathcal{R}\\
\Rightarrow&\mathcal{R}\circ \mathcal{R}\subset \mathcal{R}\\
\Leftarrow&:\\
&\forall (x,y)\in \mathcal{R},(y,z)\in \mathcal{R}\\
\Rightarrow& (x,z)\in \mathcal{R}\circ\mathcal{R},\text{又}\mathcal{R}\circ\mathcal{R}\subset\mathcal{R}\\
\Rightarrow& (x,z)\in \mathcal{R}\\
\Rightarrow&\mathcal{R}\text{具有传递性}
\end{align*}
c):\\
\begin{align*}
\Rightarrow&:\\
&\forall (x,y)\in \mathcal{R}\cap \mathcal{R}^\prime\\
\Rightarrow&(x,y)\in \mathcal{R}\land (x,y)\in \mathcal{R}^\prime\\
\Rightarrow&(x,y)\in \mathcal{R}\land (y,x)\in \mathcal{R}\text{（跟据转置关系的定义）}\\
\Rightarrow&\text{跟据反对称性}x=y,(x,y)\in \Delta_X \Rightarrow \mathcal{R}\cap \mathcal{R}^\prime \subset \Delta_X\\
\Leftarrow&:\\
&\forall (x,y)\in \mathcal{R}\land (y,x)\in \mathcal{R}\\
\Rightarrow&(x,y)\in \mathcal{R}\cap \mathcal{R}^\prime\\
\Rightarrow& (x,y)\in \Delta_X\\
\Rightarrow&x=y,\mathcal{R}\text{具有反对称性}
\end{align*}
d):\\
\begin{align*}
\Rightarrow&:\\
&\forall x\in X,y\in X,(x,y)\in X^2=\mathcal{R}\cup \mathcal{R}^\prime\\
\Rightarrow&(x,y)\in \mathcal{R}\lor(x,y)\in \mathcal{R}^\prime\\
\Leftrightarrow&(x,y)\in \mathcal{R}\lor (y,x)\in \mathcal{R}\\
\Rightarrow&\text{x,y肯定被联系起来，不考虑顺序的情况下}\\
\Leftarrow&:\\
&(x,y)\in X^2\Leftrightarrow x\in X,y\in X,\text{又}(x,y)\in \mathcal{R}\lor (y,x)\in \mathcal{R}\\
\Leftrightarrow& (x,y) \in \mathcal{R}\lor (x,y)\in \mathcal{R}^\prime\\
\Leftrightarrow&(x,y)\in \mathcal{R}\cup \mathcal{R}^\prime\\
\Rightarrow&\mathcal{R}\cup \mathcal{R}^\prime=X^2
\end{align*}
\end{solution}

\begin{exercise}.\\
    \label{def:b1-c1-s3-q2-layer}
    设$f:X\to Y$是映射。元素$y\in Y$的原像$f^{-1}(y)\subset X$称为$y$上的层\\
    \begin{align*}
    a)&:\\
    &\text{请指出映射}\mathbf{pr_1}: X_1\times X_2\to X_1,\mathbf{pr_2}: X_1\times X_2\to X_2 \text{的层}\\
    b)&:\\
    &\text{设}x_1\in X, x_2\in X,\text{如果}f(x_1)=f(x_2),\text{即如果}x_1,x_2\text{属于同一层}\\
    &\text{就认为}x_1,x_2\text{由关系}\mathcal{R}\subset X^2\text{相联系},x_1\mathcal{R}x_2\\
    &\text{请验证}:\mathcal{R}\text{是等价关系}\\
    c)&:\\
    &\text{请证明}:\text{映射}f:X\to Y\text{的层互不相交，并且所有层的并集是整个集合}X.\\
    d)&:\\
    &\text{请验证}:\text{利用集合的元素之间的任何等价关系，都可以把该集合表示为}\\
    &\text{互不相交的等价元素类的并集的形式}
    \end{align*}
\end{exercise}
\begin{solution}.\\
\label{def:b1-c1-s3-q2-brieflayer}
以下统一简写$y$上的层为$\mathbf{L} (y)$\\
a):\\
\begin{align*}
\mathbf{pr_1}&:\\
&\forall y\in X_1,\mathbf{L} (y)=\{(a,b)| (a,b)\in X_1\times X_2\land \mathbf{pr_1}((a,b))=y\}\\
\mathbf{pr_2}&:\\
&\forall y\in X_2,\mathbf{L} (y)=\{(a,b)| (a,b)\in X_1\times X_2\land \mathbf{pr_2}((a,b))=y\}
\end{align*}
b):\\
\begin{align*}
\text{自反性}&:\\
&\forall x\in X,f(x)=f(x)\\
\Rightarrow&x\mathcal{R}x\\
\text{对称性}&:\\
&\forall a,b\in X, a\mathcal{R}b\Leftrightarrow f(a)=f(b)\\
\Rightarrow& f(b)=f(a)\\
\Rightarrow& b\mathcal{R}a\\
\text{传递性}&:\\
&\forall a,b,c\in X,a\mathcal{R}b\land b\mathcal{R}c\Leftrightarrow f(a)=f(b)=f(c)\\
\Rightarrow&f(a)=f(c)\\
\Rightarrow&a\mathcal{R}c
\end{align*}
综上,$\mathcal{R}$为等价关系\\
c):\\
\begin{align*}
&\text{若}\exists y_1,y_2,(y_1\neq y_2)\land \mathbf{L}(y_1)\cap \mathbf{L}(y_2)\neq \varnothing\\
\Rightarrow & \exists x_1\in \mathbf{L}(y_1)\cap \mathbf{L}(y_2)\\
\Rightarrow &\text{由定义}: f(x_1)=y_1\land f(x_1)=y_2\land(y_1\neq y_2)\\
\Rightarrow &\text{而这于映射的单值性矛盾，因此不同的层必定不相交}\\
&\forall x_1\in X,f(x_1)\in Y,\text{假设}f(x_1)=y_1\\
\Rightarrow& x_1\in \mathbf{L}(y_1)\\
\Rightarrow&\text{所有层的并集是整个集合}X
\end{align*}
d):\\
定义等价类:$[x]=\{y|y\in X \land x\mathcal{R}y\}$\\
先证明不同的等价类一定不相交:\\
\begin{align*}
&\text{若}\exists x_1,x_2\in X,[x_1]\neq [x_2]\land [x_1]\cap [x_2]\neq \varnothing\\
\Rightarrow&\exists x_3\in X\land x_3\in  [x_1]\cap [x_2]\\
\Rightarrow&x_3\mathcal{R}x_1\land x_3\mathcal{R}x_2\\
\Rightarrow&\text{由等价关系的传递性以及对称性},x_1\mathcal{R}x_2,\text{与等价类不相同矛盾}\\
\Rightarrow&\text{不同等价类一定不相交}
\end{align*}
再证明任何一个$x\in X$均属于一个等价类:\\
显然，$\forall x_0\in X,x_0\in [x_0](\text{跟据等价关系的自反性})$\\
因此得证。
\end{solution}

\begin{exercise}
设$f:X\to Y$是$X$到$Y$的映射，请证明:\\
\begin{align*}
&\text{如果}A\text{与}B\text{是}X\text{的子集，则}:\\
&a):(A\subset B)\Rightarrow (f(A)\subset f(B))\nRightarrow (A\subset B)\\
&b):(A\neq \varnothing)\Rightarrow (f(A)\neq \varnothing)\\
&c):f(A\cap B)\subset f(A)\cap f(B)\\
&d):f(A\cup B)=f(A)\cup f(B)\\
&\text{如果}A^\prime\text{与}B^\prime\text{是}Y\text{的子集，则}:\\
&e):(A^\prime \subset B^\prime)\Rightarrow (f^{-1}(A^\prime)\subset f^{-1}(B^\prime))\\
&f):f^{-1}(A^\prime \cap B^\prime)=f^{-1}(A^\prime)\cap f^{-1}(B^\prime)\\
&g):f^{-1}(A^\prime \cup B^\prime)=f^{-1}(A^\prime)\cup f^{-1}(B^\prime)\\
&\text{如果}Y\supset A^\prime \supset B^\prime,\text{则}:\\
&h):f^{-1}(A^\prime \backslash  B^\prime )=f^{-1}(A^\prime)\backslash f^{-1}(B^\prime)\\
&i):f^{-1}(C_Y A^\prime)=C_X f^{-1}(A^\prime)\\
&\text{对于任何集合}A\subset X\text{及任何集合}B^\prime \subset Y,\text{有}:\\
&j):f^{-1}(f(A))\supset A\\
&k):f(f^{-1}(B^\prime))\subset B^\prime
\end{align*}
\end{exercise}
\begin{solution}.\\
a):\\
\begin{align*}
&A\subset B\\
\Leftrightarrow& \forall x\in A, x\in B\\
\Rightarrow& \forall y\in f(A),\exists x_0\in A,f(x_0)=y\land x_0\in B\\
\Rightarrow& \exists x_0\in B,f(x_0)=y\\
\Rightarrow& y\in f(B)\\
\Rightarrow& f(A)\subset f(B)
\end{align*}
考虑$f:\{0,1,2\}\to \{1,2\},f(0)=1,f(1)=1,f(2)=2,A=\{0\},B=\{0,1\}$\\
则可得到一个反例
b):\\
\begin{align*}
A\neq \varnothing\Rightarrow& \exists x_0\in A\\
\Rightarrow&\text{由映射定义},f(x_0)\in Y;f(x_0)\in f(A)\\
\Rightarrow&f(A)\neq \varnothing
\end{align*}
c):\\
\begin{align*}
&\forall c\in f(A\cap B)\\
\Rightarrow& \exists x_0\in A\cap B, f(x_0)=c\\
\Rightarrow& ((x_0\in A)\land(x_0\in B))\land (f(x_0)=c)\\
\Rightarrow& ((x_0\in A)\land (f(x_0)=c))\land ((x_0\in B)\land (f(x_0)=c))\\
\Rightarrow& (c\in f(A))\land (c\in f(B))\\
\Rightarrow& c\in f(A)\cap f(B)
\end{align*}
d):\\
\begin{align*}
&y\in f(A\cup B)\\
\Leftrightarrow& \exists x_0\in A\cup B \land f(x_0)=y\\
\Leftrightarrow& \exists x_0, ((x_0\in A)\lor (x_0\in B))\land f(x_0)=y\\
\Leftrightarrow& \exists x_0, ((x_0\in A)\land f(x_0)=y)\lor((x_0\in B)\land f(x_0)=y)\\
\Leftrightarrow& (y\in f(A))\lor(y\in f(b))\\
\Leftrightarrow& y\in f(A)\cup f(B)
\end{align*}
e):\\
\begin{align*}
&A^\prime \subset B^\prime\\
\Rightarrow& \forall y\in A^\prime ,y\in B^\prime\\
&\text{又} f^{-1}(A^\prime)=\{x|f(x)\in A^\prime\}\\
\Rightarrow& \forall x\in X,f(x)\in A^\prime \to f(x)\in B^\prime\\
\Rightarrow& \forall x\in f^{-1}(A^\prime), f(x)\in B^\prime\\
\Rightarrow& x\in f^{-1}(B^\prime)\\
\Rightarrow& f^{-1}(A^\prime)\subset f^{-1}(B^\prime)
\end{align*}
f):\\
\begin{align*}
&x\in f^{-1}(A^\prime \cap B^\prime)\\
\Leftrightarrow& f(x)\in (A^\prime \cap B^\prime)\\
\Leftrightarrow& (f(x) \in A^\prime)\land (f(x)\in B^\prime)\\
\Leftrightarrow& (x\in f^{-1}(A^\prime))\land (x\in f^{-1}(B^\prime))\\
\Leftrightarrow& x\in f^{-1}(A^\prime)\cap f^{-1}(B^\prime) 
\end{align*}
\newpage
g):\\
\begin{align*}
&x\in f^{-1}(A^\prime \cup B^\prime)\\
\Leftrightarrow& f(x)\in (A^\prime \cup B^\prime)\\
\Leftrightarrow& (f(x) \in A^\prime)\lor (f(x)\in B^\prime)\\
\Leftrightarrow& (x\in f^{-1}(A^\prime))\lor (x\in f^{-1}(B^\prime))\\
\Leftrightarrow& x\in f^{-1}(A^\prime)\cup f^{-1}(B^\prime) 
\end{align*}
h):\\
\begin{align*}
&x\in f^{-1}(A^\prime \backslash B^\prime)\\
\Leftrightarrow& f(x)\in A^\prime\land \lnot( f(x)\in B^\prime)\\
\Leftrightarrow& (x\in f^{-1}(A^\prime))\land \lnot (x\in f^{-1}(B^\prime))\\
\Leftrightarrow& x\in f^{-1}(A^\prime)\backslash f^{-1}(B^\prime)
\end{align*}
i):\\
在$h)$的证明中，取$A^\prime =Y$即可，仅需要变量换名。\\
j):\\
\begin{align*}
&\forall x\in A\\
\Rightarrow& f(x)\in f(A)\\
\Rightarrow x\in f^{-1}(f(A))\\
\Rightarrow f^{-1}(f(A))\supset A
\end{align*}
k):\\
\begin{align*}
&\forall y\in f(f^{-1}(B^\prime))\\
\Rightarrow& \exists x\in f^{-1}(B^\prime),f(x)=y\\
\Rightarrow& \exists (x\in X\land f(x)\in B^\prime)\land f(x)=y\\
\Rightarrow& y\in B^\prime\\
\Rightarrow& f(f^{-1}(B^\prime))\subset B^\prime
\end{align*}
\end{solution}
